# AppArmor profile for the SVG to PNG converter app.
#
# The profile name is rewritten to the app slug by Supervisor on install, so
# the name below is only a placeholder. Self-contained: no #include, so it does
# not depend on the abstraction files shipped by the host AppArmor userspace.
#
# Model: allow the container to boot and run (s6-overlay, bashio, Node.js),
# and deny the Home Assistant data directories plus the most sensitive kernel
# interfaces. AppArmor "deny" rules take precedence over the allow rules below.
profile svg_to_png flags=(attach_disconnected,mediate_deleted) {
  capability,
  network,

  # Everything the image itself needs to boot and serve
  /** rixwklm,

  # --- Home Assistant managed data stays out of reach -----------------------
  deny /config/** rwklx,
  deny /data/** rwklx,
  deny /media/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /backup/** rwklx,
  deny /app_configs/** rwklx,
  deny /local_apps/** rwklx,
  deny /homeassistant/** rwklx,
  deny /root/** rwklx,

  # --- kernel interfaces that are never needed ------------------------------
  deny /proc/sys/** w,
  deny /proc/kcore r,
  deny /sys/kernel/** w,
  deny /dev/mem rwklx,
  deny /dev/kmem rwklx,
  deny /dev/port rwklx,
  deny /dev/kmsg rwklx,
}
