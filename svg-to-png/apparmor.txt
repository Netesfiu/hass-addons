# AppArmor profile for the SVG to PNG converter app.
#
# The profile name is rewritten to the app slug by Supervisor on install, so
# the name below is only a placeholder.
#
# Intentionally self-contained: no #include, so the profile does not depend on
# the abstraction files shipped by the AppArmor userspace on the host. It grants
# exactly what a Node.js HTTP service needs (read its own code, resolve DNS,
# speak TCP, write temp files) and denies the Home Assistant data directories.
profile svg_to_png flags=(attach_disconnected,mediate_deleted) {
  # --- networking -----------------------------------------------------------
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # --- runtime binary and libraries -----------------------------------------
  /usr/bin/node ix,
  /usr/lib/** mr,
  /lib/** mr,
  /usr/share/** r,

  # --- application code -----------------------------------------------------
  /app/** rm,

  # --- TLS, resolver, system info -------------------------------------------
  /etc/ssl/** r,
  /etc/ca-certificates/** r,
  /etc/resolv.conf r,
  /etc/hosts r,
  /etc/services r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/group r,
  /etc/localtime r,
  /proc/** r,
  /sys/devices/system/cpu/** r,

  # --- devices and scratch space --------------------------------------------
  /dev/null rw,
  /dev/zero r,
  /dev/urandom r,
  /dev/random r,
  /tmp/** rwkl,
  /run/** rwkl,

  # --- Home Assistant managed data stays out of reach ------------------------
  deny /config/** rwklx,
  deny /data/** rwklx,
  deny /media/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /backup/** rwklx,
  deny /app_configs/** rwklx,
  deny /local_apps/** rwklx,
  deny /homeassistant/** rwklx,
  deny /root/** rwklx,
  deny /proc/sys/** rwklx,
  deny /sys/kernel/** rwklx,
  deny /dev/mem rwklx,
  deny /dev/kmem rwklx,
  deny /dev/port rwklx,
}
