# AppArmor profile for the SVG to PNG converter app.
#
# The profile name is rewritten to the app slug by Supervisor on install, so
# the name below is only a placeholder.
#
# Intentionally self-contained (no #include) so it does not depend on the
# abstraction files shipped by the host AppArmor userspace.
#
# Model: the app (Alpine + s6-overlay + bashio, then Node.js) may read and
# execute what it needs to boot and serve; the Home Assistant data directories
# and the sensitive /proc and /sys areas are explicitly denied.
profile svg_to_png flags=(attach_disconnected,mediate_deleted) {
  # --- capabilities ---------------------------------------------------------
  capability,

  # --- networking -----------------------------------------------------------
  network,

  # --- boot: s6-overlay, bashio, shells and tooling -------------------------
  /init ix,
  /bin/** rix,
  /sbin/** rix,
  /usr/** rix,
  /lib/** rm,
  /package/** rix,
  /etc/cont-init.d/** rix,
  /etc/services.d/** rix,
  /etc/fix-attrs.d/** r,
  /etc/s6/** r,
  /etc/s6-overlay/** rix,
  /etc/periodic/** rix,

  # --- configuration, TLS, resolver ----------------------------------------
  /etc/** r,
  /run/** rwkl,
  /var/run/** rwkl,
  /var/log/** rwkl,
  /var/lib/** rwkl,
  /tmp/** rwkl,
  /app/** r,
  /proc/** rwk,
  /sys/** r,

  # --- devices --------------------------------------------------------------
  /dev/** rwkl,
  deny /dev/mem rwklx,
  deny /dev/kmem rwklx,
  deny /dev/port rwklx,

  # --- Home Assistant managed data stays out of reach -----------------------
  deny /config/** rwklx,
  deny /data/** rwklx,
  deny /media/** rwklx,
  deny /share/** rwklx,
  deny /ssl/** rwklx,
  deny /backup/** rwklx,
  deny /app_configs/** rwklx,
  deny /local_apps/** rwklx,
  deny /homeassistant/** rwklx,
  deny /root/** rwklx,
  deny /proc/sys/** w,
  deny /proc/kcore r,
  deny /sys/kernel/** w,
}
